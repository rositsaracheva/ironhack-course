# Python / PySpark (Board Notes)
file1=sc.parallelize("good sentence, good day , good boy)
file_split = file1.flatMap(lambda s:split(" ")
file_kv = file_split.map(lambda w:(w,1))
file_wc = file_kv.reduceByKey( lambda a,b :(a+b))

# --- PySpark Terminal Execution & Output (pyspark) ---
>>> file1 = sc.parallelize(["good sentence, good day , good boy"])
>>> file_split = file1.flatMap(lambda s: s.split(" "))
>>> file_kv = file_split.map(lambda w: (w, 1))
>>> file_wc = file_kv.reduceByKey(lambda a, b: (a + b))
>>> file_wc.collect()
[('good', 3), ('day', 1), (',', 1), ('boy', 1), ('sentence,', 1)]


================================================================================
PART 2: 1. FIND HOW TO READ A TEXTFILE FROM LOCAL IN BOTH PYSPARK AND SCALA
================================================================================
* Tip: Use `"sample.txt"` (or `"/full/path/sample.txt"`) in `local[*]` mode, or prefix
  with `"file:///full/path/sample.txt"` to explicitly force local filesystem reading.

--------------------------------------------------------------------------------
1. SCALA (`spark-shell`)
--------------------------------------------------------------------------------
# Method A: As an RDD (sc.textFile)
scala> val file1 = sc.textFile("sample.txt")
val file1: org.apache.spark.rdd.RDD[String] = sample.txt MapPartitionsRDD[1] at textFile at <console>:1

scala> file1.collect()
val res0: Array[String] = Array(good sentence, good day , good boy, spark is fast and spark is easy)

# Method B: As a DataFrame (spark.read.text)
scala> val df = spark.read.text("sample.txt")
val df: org.apache.spark.sql.DataFrame = [value: string]

scala> df.show(truncate = false)
+----------------------------------+
|value                             |
+----------------------------------+
|good sentence, good day , good boy|
|spark is fast and spark is easy   |
+----------------------------------+


--------------------------------------------------------------------------------
2. PYSPARK (`pyspark`)
--------------------------------------------------------------------------------
# Method A: As an RDD (sc.textFile)
>>> file1 = sc.textFile("sample.txt")
>>> file1.collect()
['good sentence, good day , good boy', 'spark is fast and spark is easy']

# Method B: As a DataFrame (spark.read.text)
>>> df = spark.read.text("sample.txt")
>>> df.show(truncate=False)
+----------------------------------+
|value                             |
+----------------------------------+
|good sentence, good day , good boy|
|spark is fast and spark is easy   |
+----------------------------------+
